# IP 通用演员化 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 IP 从固定职能的吉祥物变为可适配不同场景的通用演员——LLM 在视觉身份硬约束下自由决定每帧的角色、服装、动作、表情、构图和出场大小。

**Architecture:** 三层分离。Layer 1 视觉身份（不变），Layer 2 角色职能（LLM 按帧决定），Layer 3 出场规格（LLM 按帧决定）。IPProfile 新增 `role_presets` / `presence_spectrum` / `universal_constraints`。IPUsagePlanner 从确定性拼接改为 LLM 驱动的帧级自然语言描述生成。IPFrameAdaptationPackage 现有字段（outfit_theme / accessories / pose / expression 等）被正式填充。

**Tech Stack:** Python dataclasses, Pydantic v2, FastAPI, Streamlit component helpers, pytest, LLM API client (existing infrastructure), local JSON repositories, Z-Image prompt assembly.

class IPFrameAppearancePlanner:
    """LLM-driven per-frame IP appearance planner.
    
    Extends the deterministic IPUsagePlanner by adding LLM-generated
    natural-language appearance descriptions and filling outfit/accessory/
    pose/expression fields in IPFrameAdaptationPackage.
    """
    
    def __init__(self, *, llm_client=None):
        self._deterministic = IPUsagePlanner()
        self._llm = llm_client
    
    def plan_batch(self, *, storyboard_plan, ip_profile, resolved_style=None,
                   scene_casts_by_frame=None, generation_world_profile=None,
                   previous_frame_appearance=None):
        # 1. Run deterministic planner for presence_type baseline
        base_packages = self._deterministic.plan_batch(
            storyboard_plan=storyboard_plan,
            ip_profile=ip_profile,
            resolved_style=resolved_style,
            scene_casts_by_frame=scene_casts_by_frame,
            generation_world_profile=generation_world_profile,
        )
        
        # 2. For each frame, generate LLM appearance description
        enriched = []
        prev_appearance = previous_frame_appearance
        
        for i, (frame, base) in enumerate(zip(storyboard_plan.frames, base_packages)):
            if base.ip_presence_type == IPPresenceType.ABSENT:
                enriched.append(base)
                prev_appearance = None
                continue
            
            appearance = self._plan_frame_appearance(
                frame=frame,
                ip_profile=ip_profile,
                base_package=base,
                previous_appearance=prev_appearance,
            )
            prev_appearance = appearance.get("appearance_description")
            
            enriched.append(self._merge_appearance(base, appearance))
        
        return enriched
    
    def _plan_frame_appearance(self, *, frame, ip_profile, base_package, previous_appearance):
        """Call LLM to decide role, outfit, action, expression, composition."""
        if self._llm is None:
            return self._rule_based_appearance(frame, ip_profile, base_package)
        return self._llm_driven_appearance(
            frame=frame,
            ip_profile=ip_profile,
            base_package=base_package,
            previous_appearance=previous_appearance,
        )
    
    def _rule_based_appearance(self, frame, ip_profile, base_package):
        """Deterministic fallback when no LLM client available.
        
        Uses variable_slots and role_presets to create varied descriptions
        based on frame content domain keywords.
        """
        return {
            "appearance_description": _build_rule_based_description(
                frame, ip_profile, base_package
            ),
            "outfit_theme": _infer_outfit_from_content(frame, ip_profile),
            "action": base_package.action,
            "expression": base_package.expression,
            "pose": base_package.pose,
            "accessories": _infer_accessories_from_content(frame, ip_profile),
        }
    
    def _llm_driven_appearance(self, *, frame, ip_profile, base_package, previous_appearance):
        """Build LLM prompt and parse response."""
        prompt = _build_appearance_prompt(
            frame=frame,
            ip_profile=ip_profile,
            base_package=base_package,
            previous_appearance=previous_appearance,
        )
        response = self._llm.complete(prompt)  # structured JSON output
        return _parse_appearance_response(response)
    
    def _merge_appearance(self, base_package, appearance):
        """Create enriched IPFrameAdaptationPackage with LLM-generated fields."""
        return IPFrameAdaptationPackage(
            frame_id=base_package.frame_id,
            ip_presence_type=base_package.ip_presence_type,
            presence_mode=base_package.presence_mode,
            semantic_reason=base_package.semantic_reason,
            must_not_replace=base_package.must_not_replace,
            identity_anchors_visible=base_package.identity_anchors_visible,
            identity_anchors_suppressed=base_package.identity_anchors_suppressed,
            identity_color_terms=base_package.identity_color_terms,
            outfit_theme=appearance.get("outfit_theme"),
            outfit_condition=appearance.get("outfit_condition"),
            accessories=tuple(appearance.get("accessories", ())),
            action=appearance.get("action") or base_package.action,
            expression=appearance.get("expression") or base_package.expression,
            pose=appearance.get("pose") or base_package.pose,
            camera_relationship=base_package.camera_relationship,
            depth_layer=base_package.depth_layer,
            interaction_target=appearance.get("interaction_target") or base_package.interaction_target,
            continuity_from_previous=appearance.get("continuity_from_previous"),
            shot_fit_notes=base_package.shot_fit_notes,
            image_text_plan=base_package.image_text_plan,
            prompt_weight=base_package.prompt_weight,
            negative_constraints=base_package.negative_constraints,
        )


def _build_appearance_prompt(*, frame, ip_profile, base_package, previous_appearance):
    """Build the LLM prompt for per-frame IP appearance design."""

def _parse_appearance_response(response: str) -> dict:
    """Parse LLM JSON response into appearance dict."""
    import json
    try:
        # Extract JSON from possible markdown code block
        if "```" in response:
            response = response.split("```")[1]
            if response.startswith("json"):
                response = response[4:]
        data = json.loads(response.strip())
        return {
            "appearance_description": data.get("appearance_description"),
            "outfit_theme": data.get("outfit_theme"),
            "outfit_condition": None,
            "accessories": data.get("accessories", []),
            "action": data.get("action"),
            "expression": data.get("expression"),
            "pose": data.get("pose"),
            "interaction_target": data.get("interaction_target"),
            "continuity_from_previous": data.get("continuity_from_previous"),
        }
    except (json.JSONDecodeError, IndexError, KeyError):
        return {}


def _build_rule_based_description(frame, ip_profile, base_package):
    """Deterministic fallback: build appearance description from variable_slots + content domain."""
    # This is the no-LLM fallback that still provides variation
    # based on detecting content domain keywords
    identity_str = "、".join(ip_profile.identity_lock)
    presence = base_package.presence_mode or "guide"
    
    domain = _detect_content_domain(frame)
    
    templates = {
        "文旅": f"{identity_str}站在场景中，做温和的讲解手势，半身中景，面向画面内容",
        "爱情": f"{identity_str}在画面边缘，安静陪伴，局部出镜或远景，不抢主体",
        "美食": f"{identity_str}坐在餐桌旁，好奇地看着食物，半身或远景",
        "科技": f"{identity_str}在背景中观察，远景融入，保持低调",
        "日常": f"{identity_str}自然地融入场景，随画面内容调整动作和表情",
        "通用": f"{identity_str}根据画面内容自然地出现，保持角色可识别",
    }
    
    return templates.get(domain, templates["通用"])


def _detect_content_domain(frame) -> str:
    """Detect content domain from frame text for rule-based fallback."""
    text = _frame_text(frame)
    domains = {
        "文旅": ("古城", "城墙", "寺庙", "景点", "旅游", "导览", "游客", "长乐门", "正定"),
        "爱情": ("爱情", "情侣", "感情", "喜欢", "爱", "在一起", "分手", "思念"),
        "美食": ("美食", "食物", "吃", "餐厅", "厨房", "料理"),
        "科技": ("科技", "AI", "数据", "代码", "电脑", "技术"),
        "日常": ("生活", "家", "工作", "学习", "城市", "街道"),
    }
    for domain, keywords in domains.items():
        if any(kw in text for kw in keywords):
            return domain
    return "通用"


def _infer_outfit_from_content(frame, ip_profile) -> str | None:
    """Infer outfit theme from content keywords."""
    text = _frame_text(frame)
    if any(kw in text for kw in ("古城", "历史", "文旅", "寺庙")):
        return "清爽文旅风格服装"
    if any(kw in text for kw in ("爱情", "情侣", "感情")):
        return "温暖日常服装"
    if any(kw in text for kw in ("美食", "厨房", "餐厅")):
        return "休闲餐饮风格服装"
    return None


def _infer_accessories_from_content(frame, ip_profile) -> tuple[str, ...]:
    """Infer accessories from content keywords."""
    text = _frame_text(frame)
    items = []
    if any(kw in text for kw in ("导览", "旅游", "古城", "景点")):
        items.append("导览旗或地图")
    if any(kw in text for kw in ("爱情", "情侣", "感情")):
        items.append("花束或咖啡杯")
    if any(kw in text for kw in ("美食", "厨房", "餐厅")):
        items.append("餐具或菜单")
    return tuple(items)
```
