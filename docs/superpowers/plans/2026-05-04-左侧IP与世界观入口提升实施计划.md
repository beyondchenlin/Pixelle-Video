# 左侧 IP 与世界观入口提升 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将标准生成页的 IP 选择与本次世界观提示提升到左侧内容输入区，成为与分镜规划同层级的一等入口。

**Architecture:** 复用现有 IP AssetBible 客户端、世界观草稿 API、`generation_world_hint` 合同和正式生成请求构建链路。新增左侧 `content_ip_world_controls` 模块作为权威 UI 状态源；高级分镜只保留镜头/预设/连续性职责；中间风格区不再渲染标准生成 IP 主入口，避免两套选择器冲突。

**Tech Stack:** Streamlit UI、Python pytest、现有 `web.components` 组件、现有 `web.utils.content_api.generate_world_hint_draft`、现有 `IPControlsContract` / `StoryboardControlsContract`。

        self.button_calls = []
        self.warning_calls = []

    def expander(self, label, expanded=False):
        self.expanders.append({"label": label, "expanded": expanded})
        return _FakeContext()

    def toggle(self, label, value=False, **kwargs):
        key = kwargs.get("key")
        self.toggle_calls.append({"label": label, "value": value, **kwargs})
        return bool(self.session_state.get(key, value))

    def selectbox(self, label, options, index=0, key=None, **kwargs):

```python
from __future__ import annotations

import logging
from collections.abc import Callable, Mapping, Sequence
from typing import Any

import streamlit as st

from pixelle_video.prompt_language import CHINESE_PROMPT_LANGUAGE
from web.components.ip_prompt_chain_controls import (
    load_ip_prompt_chain_asset_bibles,
    render_ip_prompt_chain_controls,
)
from web.i18n import tr
from web.utils.content_api import generate_world_hint_draft
from web.utils.streamlit_helpers import safe_rerun

logger = logging.getLogger(__name__)

CONTENT_IP_STATE_PREFIX = "content_ip"
CONTENT_WORLD_HINT_KEY = "content_generation_world_hint"
CONTENT_WORLD_HINT_SOURCE_KEY = "content_generation_world_hint_source"
CONTENT_IP_PROFILE_WORLD_HINT_KEY = "content_ip_profile_world_hint"


def build_content_ip_world_payload(
    *,
    ip_payload: Mapping[str, Any],
    generation_world_hint: str | None,
) -> dict[str, Any]:
    payload = {
        key: ip_payload[key]
        for key in ("ip_enabled", "ip_asset_bible_id", "ip_profile_id")
        if key in ip_payload
    }
    if "ip_enabled" not in payload:
        payload["ip_enabled"] = False
    world_hint = str(generation_world_hint or "").strip()
    if world_hint:
        payload["generation_world_hint"] = world_hint
    return payload


def render_content_ip_world_controls(
    *,
    pixelle_video,
    content_context: Mapping[str, object] | None = None,
    storyboard_prompt_language: str = CHINESE_PROMPT_LANGUAGE,
    world_preset_id: str | None = None,
    ui=st,
    translate: Callable[..., str] = tr,
    asset_bible_loader: Callable[[], Sequence[Mapping[str, Any]]] | None = None,
    world_hint_draft_generator=None,
) -> dict[str, Any]:
    """Render the left-column IP and request-scoped world-hint controls."""
    with ui.expander(translate("content.ip_world.section_title"), expanded=True):
        loader = asset_bible_loader or (
            lambda: _load_content_ip_asset_bibles(
                pixelle_video=pixelle_video,
                session_state=ui.session_state,
                ui=ui,
                translate=translate,
            )
        )
        ip_payload = dict(
            render_ip_prompt_chain_controls(
                ui=ui,
                asset_bible_loader=loader,
                translate=translate,
                state_key_prefix=CONTENT_IP_STATE_PREFIX,
                label_key_prefix="content.ip_world",
            )
        )
        ip_profile_world_hint = str(ip_payload.pop("ip_profile_world_hint", "")).strip()
        if ip_profile_world_hint:
            ui.session_state[CONTENT_IP_PROFILE_WORLD_HINT_KEY] = ip_profile_world_hint
        else:
            ui.session_state.pop(CONTENT_IP_PROFILE_WORLD_HINT_KEY, None)

        current_world_hint = ui.text_area(
            translate("content.ip_world.generation_world_hint"),
            key=CONTENT_WORLD_HINT_KEY,
            height=96,
            help=translate("content.ip_world.generation_world_hint_help"),
            on_change=_mark_world_hint_manual,
            kwargs={"session_state": ui.session_state},
        )

        draft_generator = world_hint_draft_generator or generate_world_hint_draft
        action_col1, action_col2 = ui.columns(2)
        with action_col1:
            if ui.button(
                translate("content.ip_world.generate_from_content"),
                key="content_world_hint_generate_from_content",
                width="stretch",
            ):
                source_text = str((content_context or {}).get("text") or "").strip()
                if not source_text:
                    ui.warning(translate("content.ip_world.missing_content"))
                else:
                    response = draft_generator(
                        source_text=source_text,
                        title=str((content_context or {}).get("title") or "").strip() or None,
                        world_preset_id=world_preset_id,
                        storyboard_prompt_language=storyboard_prompt_language,
                        ip_default_world_hint=ip_profile_world_hint or None,
                    )
                    ui.session_state[CONTENT_WORLD_HINT_KEY] = str(
                        response.get("world_hint_draft") or ""
                    ).strip()
                    ui.session_state[CONTENT_WORLD_HINT_SOURCE_KEY] = "generated_from_script"
                    safe_rerun()
        with action_col2:
            if ui.button(
                translate("content.ip_world.use_ip_default"),
                key="content_world_hint_use_ip_default",
                width="stretch",
            ):
                if not ip_profile_world_hint:
                    ui.warning(translate("content.ip_world.missing_ip_default"))
                else:
                    ui.session_state[CONTENT_WORLD_HINT_KEY] = ip_profile_world_hint
                    ui.session_state[CONTENT_WORLD_HINT_SOURCE_KEY] = "ip_default"
                    safe_rerun()

    return build_content_ip_world_payload(
        ip_payload=ip_payload,
        generation_world_hint=ui.session_state.get(CONTENT_WORLD_HINT_KEY, current_world_hint),
    )


def _load_content_ip_asset_bibles(*, pixelle_video, session_state, ui, translate) -> list[dict[str, Any]]:
    if pixelle_video is None:
        return []
    try:
        return load_ip_prompt_chain_asset_bibles(
            pixelle_video=pixelle_video,
            session_state=session_state,
        )
    except Exception:
        logger.exception("failed to load content IP asset bibles")
        ui.warning(translate("content.ip_world.load_failed"))
        return []


def _mark_world_hint_manual(*, session_state) -> None:
    session_state[CONTENT_WORLD_HINT_SOURCE_KEY] = "manual"


__all__ = [
    "CONTENT_IP_PROFILE_WORLD_HINT_KEY",
    "CONTENT_IP_STATE_PREFIX",
    "CONTENT_WORLD_HINT_KEY",
    "CONTENT_WORLD_HINT_SOURCE_KEY",
    "build_content_ip_world_payload",
    "render_content_ip_world_controls",
]
```
